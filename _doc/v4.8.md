## HOOK 位置

### IN_CLOSE_WRITE / IN_CLOSE_NOWRITE — `File::close()`

```rust
FileType::Inode { inode } | FileType::Device { inode, .. } => {
    let mask = if inner_copy.writeable {
        IN_CLOSE_WRITE   // 檔案以寫入模式開啟後關閉
    } else {
        IN_CLOSE_NOWRITE // 檔案以唯讀模式開啟後關閉
    };
    inotify::notify(inode.dev, inode.inum, mask, 0, "");
    // ... put inode
}
```

### IN_ACCESS — `File::read()` for Inode

```rust
FileType::Inode { inode } => {
    let (dev, inum) = (inode.dev, inode.inum);
    // ... do actual read ...
    if let Ok(read) = read {
        inotify::notify(dev, inum, IN_ACCESS, 0, "");
        Ok(read as usize)
    }
}
```
